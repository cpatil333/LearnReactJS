import "./App.css";
import Home from "./E-CommerceProject/Home";
import Products from "./E-CommerceProject/Products";
import SingleProduct from "./E-CommerceProject/SingleProduct";
import { createBrowserRouter, RouterProvider } from "react-router-dom";
import Layout from "./E-CommerceProject/Layout";

function App() {
  const router = createBrowserRouter([
    {
      path: "/",
      element: <Layout />,
      children: [
        { path: "/", element: <Home /> },
        { path: "/products", element: <Products /> },
        { path: "/products/:id", element: <SingleProduct /> },
      ],
    },
  ]);
  return (
    <>
      <RouterProvider router={router} />
    </>
  );
}

export default App;
